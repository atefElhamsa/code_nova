## 💻 التطبيق المعماري (MVVM & Clean Code)

### 1. كتابة الاختبار النهائي للـ ViewModel

```dart
// ==========================================
// File: test/features/tasks/presentation/viewmodels/task_list_viewmodel_test.dart
// Description: اختبارات الشمولية والتصفية لمشروع Capstone
// Architecture Layer: Unit Tests
// ==========================================

import 'package:flutter_test/flutter_test.dart';
import 'package:mocktail/mocktail.dart';

import 'package:doc_flutter/features/tasks/data/models/task_model.dart';
import 'package:doc_flutter/features/tasks/data/repositories/task_repository_impl.dart';
import 'package:doc_flutter/features/tasks/presentation/viewmodels/task_list_viewmodel.dart';
import 'package:doc_flutter/core/enums/view_status.dart';

class MockTaskRepository extends Mock implements BaseTaskRepository {}

void main() {
  late TaskListViewModel viewModel;
  late MockTaskRepository mockRepository;

  setUp(() {
    mockRepository = MockTaskRepository();
    viewModel = TaskListViewModel(mockRepository);
  });

  const tTasksList = [
    TaskModel(id: '1', title: 'مهمة 1', description: 'تطبيق MVVM', isCompleted: false),
    TaskModel(id: '2', title: 'مهمة 2', description: 'كتابة الاختبارات', isCompleted: true),
  ];

  group('اختبارات TaskListViewModel الشاملة -', () {
    test('يجب تصفية المهام المكتملة فقط عند اختيار TaskFilter.completed', () async {
      // Arrange
      when(() => mockRepository.getTasks()).thenAnswer((_) async => tTasksList);

      // Act
      await viewModel.fetchTasks();
      viewModel.setFilter(TaskFilter.completed);

      // Assert
      expect(viewModel.status, ViewStatus.success);
      expect(viewModel.filteredTasks.length, 1);
      expect(viewModel.filteredTasks.first.id, '2');
    });
  });
}
```
